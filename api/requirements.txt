# Runtime dependencies -- FULLY PINNED, including transitive ones.
#
# WHY EXACT PINS. This file previously carried floors only (`fastapi>=0.110`),
# which meant every `docker build` resolved to whatever was newest that day.
# The image that booted in July was not the image that booted in September, and
# nothing recorded the difference. The drift was not hypothetical: with
# `fastapi>=0.110` declared, the resolved version was 0.139.2, and starlette had
# gone through a major release to 1.3.1 underneath it.
#
# For a project meant to run for years, a build that cannot be reproduced is a
# build that cannot be rolled back. Upgrades belong in a commit, where they can
# be tested and reverted, not in whatever pip happened to fetch.
#
# Every version below is the one the 678-test suite passes against.
#
# LAYOUT. Direct dependencies first, each with the reason it is here.
# Transitive pins follow, annotated with what pulled them in -- so a future
# upgrade can tell what is safe to touch from what is merely along for the ride.
#
# TO UPGRADE. Change a direct pin, then re-resolve the transitive block:
#
#     pip install pip-tools
#     pip-compile --upgrade-package fastapi requirements.txt
#
# or, without pip-tools, install into a clean venv and `pip freeze`. Either way,
# run `python -m pytest tests/` before committing. Do it deliberately and on
# purpose -- that is the entire point of this file.
#
# Test-only dependencies are NOT here; see requirements-dev.txt. Keeping them
# out means pytest and its tree stay off the production image.

# Multipart form parsing, which FastAPI needs for UploadFile. The chart
# screenshot attached to a plan (POST /api/plans/{id}/chart) is the only
# file upload in the app; without this FastAPI raises at import time.
python-multipart==0.0.32

# Clerk JWT verification (api/auth.py). The [crypto] extra pulls in
# `cryptography`, which PyJWT needs to check RS256 signatures.
pyjwt[crypto]==2.13.0

# The IANA timezone database, for `ZoneInfo("America/New_York")`.
#
# Nothing declared this before, and the omission was invisible for the worst
# possible reason: it was satisfied by accident everywhere it was tested. On a
# dev machine some other package had already pulled tzdata in; in Docker,
# python:3.12-slim happens to carry /usr/share/zoneinfo, which zoneinfo reads
# before falling back to this package. Neither is a guarantee, and a clean
# install with --no-deps found it immediately.
#
# The failure mode is not a bad response, it is no service at all. MARKET_TZ is
# constructed at module import time in main.py, services/analytics.py and
# services/ibkr_parser.py, so a missing tz database raises ZoneInfoNotFoundError
# during import and the container dies at boot rather than at a request.
#
# Declaring it also pins the tz rules themselves. DST boundaries decide which
# session a fill is classified into, so "America/New_York" quietly changing
# under a rebuild is a data-correctness event; this makes it a commit.
tzdata==2026.3

# Production WSGI/ASGI process manager. Previously installed by a bare
# `pip install gunicorn` line in the Dockerfile, which put the one process that
# supervises everything else outside the lockfile entirely.
gunicorn==26.0.0

# uvicorn[standard]'s event loop, and the reason a naive `pip freeze` on Windows
# produces a lockfile that is wrong for the Docker image: uvloop does not build
# there, so it is absent from a Windows environment and would be silently
# dropped from the lock. The marker is uvicorn's own, reproduced here so the pin
# applies exactly where the dependency does.
uvloop==0.22.1 ; sys_platform != 'win32' and sys_platform != 'cygwin' and platform_python_implementation != 'PyPy'

# The mirror image: click needs colorama to render ANSI on Windows, so a lock
# generated on linux would leave a dev machine's install unpinned. Pinned here
# so both platforms resolve to a fully specified set.
colorama==0.4.6 ; platform_system == 'Windows'
